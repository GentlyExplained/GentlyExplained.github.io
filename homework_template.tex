\begin{problem}{1}{Let $A \subset \mathbb{R}$ be nonempty and bounded above. Prove that if $M$ is an upper bound of $A$ and $M \in A$, then $M = \sup A$.}
\begin{proof}
    Suppose $M$ is an upper bound of $A$ and $M \in A$. 
    By definition of an upper bound, $a \le M$ for all $a \in A$.
    Now let $M'$ be any arbitrary upper bound of $A$. 
    Since $M \in A$, it must follow by definition that $M \le M'$.
    Thus, $M$ is less than or equal to every other upper bound, making it the least upper bound.
    Therefore, $M = \sup A$.
\end{proof}
\end{problem}
